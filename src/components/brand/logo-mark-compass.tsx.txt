import { cn } from '@/lib/utils';

const CX = 334, CY = 338;
const rad = (d: number) => (d * Math.PI) / 180;
const pt = (r: number, deg: number) => `${(CX + r * Math.cos(rad(deg))).toFixed(1)} ${(CY + r * Math.sin(rad(deg))).toFixed(1)}`;
/** Төвөөс (CX,CY) r радиустай нум: a0 → a1 градус (цагийн зүүгээр, 0° = баруун) */
const arc = (r: number, a0: number, a1: number) => `M ${pt(r, a0)} A ${r} ${r} 0 ${a1 - a0 > 180 ? 1 : 0} 1 ${pt(r, a1)}`;

/**
 * «Чиглэл» брэндийн луужин тэмдэг (SVG). Хар хөх хэсэг нь currentColor биш, `--foreground` токенийг дагана
 * (Dark горимд автоматаар цайвар болно); Teal нь тогтмол; голын нүх нь картын өнгө.
 */
export function LogoMark({ className, title }: { className?: string; title?: string }) {
  return (
    <svg viewBox="0 0 668 676" role={title ? 'img' : undefined} aria-label={title} aria-hidden={title ? undefined : true} className={cn('shrink-0', className)}>
      <g fill="var(--foreground)">
        <polygon points="334,4 312,80 356,80" /><polygon points="334,672 312,596 356,596" />
        <polygon points="2,338 78,316 78,360" /><polygon points="666,338 590,316 590,360" />
        <polygon points="297,300 376,377 152,524" />
      </g>
      <g fill="none" stroke="var(--foreground)" strokeWidth={30}>
        <path d={arc(236, -41, 121)} /><path d={arc(236, 139, 301)} />
      </g>
      <path d={arc(186, 0, 121)} fill="none" stroke="var(--foreground)" strokeWidth={13} />
      <g fill="none" stroke="var(--logo-teal)" strokeWidth={13}>
        <path d={arc(190, 180, 270)} /><path d={arc(155, 180, 270)} />
      </g>
      <polygon points="518,152 297,298 376,377" fill="var(--logo-teal)" />
      <circle cx={CX} cy={CY} r={29} fill="var(--card)" />
    </svg>
  );
}
